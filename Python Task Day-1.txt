Q1. Temperature Conversion & Formatted Output

c = float(input("Enter Celsius: "))
f = (c * 9/5) + 32
k = c + 273.15
print(f"Celsius: {c:.2f}°C")
print(f"Fahrenheit: {f:.2f}°F")
print(f"Kelvin: {k:.2f} K")
print("\nConversion Table")
print("Celsius  Fahrenheit  Kelvin")
for c in range(-40, 101, 10):
    f = (c * 9/5) + 32
    k = c + 273.15
    print(f"{c:7.2f} {f:11.2f} {k:8.2f}")

Q4. Placement Bill Calculator

items = []
total = 0
while True:
    x = input("Enter price or done: ")
    if x.lower() == "done":
        break
    price = float(x)
    items.append(price)
    total += price
if total >= 500:
    discount = total * 0.10
else:
    discount = total * 0.05
after_discount = total - discount
tax = after_discount * 0.05
final = after_discount + tax
print("\nItem        Price")
print("------------------")
for i, price in enumerate(items, 1):
    print(f"Item {i:<5} {price:.2f}")
print("------------------")
print(f"Subtotal     {total:.2f}")
print(f"Discount     {discount:.2f}")
print(f"Tax          {tax:.2f}")
print(f"Final        {final:.2f}")

Q6. Simple Interest Report

p = float(input("Principal: "))
r = float(input("Rate: "))
t = float(input("Time: "))
if p < 0 or t < 0:
    print("Invalid principal or time")
else:
    si = (p * r * t) / 100
    total = p + si
    print(f"Principal : {p:.2f}")
    print(f"Rate      : {r:.2f}%")
    print(f"Time      : {t:.2f} years")
    print(f"Simple Interest: {si:.2f}")
    print(f"Total Amount : {total:.2f}")
