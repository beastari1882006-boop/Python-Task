Q1. E1 — Prime, Factorial, Fibonacci, Digit Sum & Reverse

N = int(input("Enter N: "))
num = int(input("Enter Number: "))
print("Primes:", end=" ")
for n in range(2, N + 1):
    prime = True
    for i in range(2, n):
        if n % i == 0:
            prime = False
            break
    if prime:
        print(n, end=" ")
fact = 1
for i in range(1, N + 1):
    fact *= i
print("\nFactorial:", fact)
a, b = 0, 1
print("Fibonacci:", end=" ")
for i in range(N):
    print(a, end=" ")
    a, b = b, a + b
temp = num
digit_sum = 0
reverse = 0
while temp > 0:
    digit = temp % 10
    digit_sum += digit
    reverse = reverse * 10 + digit
    temp //= 10
print("\nDigit Sum:", digit_sum)
print("Reverse:", reverse)


Q2. E2 — Patterns & Multiplication Grid


N = int(input("Enter N: "))
print("\nRight Triangle")
for i in range(1, N + 1):
    for j in range(i):
        print("*", end="")
    print()
print("\nPyramid")
for i in range(1, N + 1):
    for j in range(N - i):
        print(" ", end="")
    for j in range(2 * i - 1):
        print("*", end="")
    print()
print("\nNumber Triangle")
for i in range(1, N + 1):
    for j in range(1, i + 1):
        print(j, end="")
    print()
print("\nMultiplication Grid")
for i in range(1, 11):
    for j in range(1, 11):
        print(f"{i} x {j} = {i*j}", end="    ")
    print()


Q3. E3 — String Analyzer


text = input("Enter a string: ")
reverse = ""
for ch in text:
    reverse = ch + reverse
slice_reverse = text[::-1]
clean = ""
for ch in text:
    if ch != " ":
        clean += ch.lower()
palindrome = clean == clean[::-1]
vowels = 0
consonants = 0
digits = 0
for ch in text:
    if ch.isdigit():
        digits += 1
    elif ch.isalpha():
        if ch.lower() in "aeiou":
            vowels += 1
        else:
            consonants += 1
print("Loop Reverse:", reverse)
print("Slice Reverse:", slice_reverse)
print("Palindrome:", "Yes" if palindrome else "No")
print("Vowels:", vowels)
print("Consonants:", consonants)
print("Digits:", digits)


Q4. Formatted Electricity Bill


customer = input("Customer: ")
units = int(input("Units: "))
if units <= 100:
    amount = units * 2
elif units <= 200:
    amount = 100 * 2 + (units - 100) * 3
elif units <= 300:
    amount = 100 * 2 + 100 * 3 + (units - 200) * 4
else:
    amount = 100 * 2 + 100 * 3 + 100 * 4 + (units - 300) * 5
print(f"Customer : {customer}")
print(f"Units    : {units}")
print(f"Amount   : ₹{amount:.2f}")


Q5. ATM Cash Withdrawal & Denomination Breakdown


amount = int(input("Withdrawal Amount: "))
if amount <= 0 or amount > 20000 or amount % 10 != 0:
    print("Invalid withdrawal amount")
else:
    remaining = amount
    total_notes = 0
    for denomination in [500, 200, 100, 50, 10]:
        notes = remaining // denomination
        remaining %= denomination
        total_notes += notes
        print(f"₹{denomination} notes : {notes}")
    print("Total Notes :", total_notes)
    print(f"Amount      : ₹{amount}")


Q6. Digital Parking Fee & Time Calculator


entry = input("Entry Time: ")
exit_time = input("Exit Time: ")
eh, em = map(int, entry.split(":"))
xh, xm = map(int, exit_time.split(":"))
entry_minutes = eh * 60 + em
exit_minutes = xh * 60 + xm
if exit_minutes < entry_minutes:
    exit_minutes += 24 * 60
duration = exit_minutes - entry_minutes
hours = duration // 60
minutes = duration % 60
billable_hours = (duration + 59) // 60
if billable_hours == 0:
    fee = 0
else:
    fee = 30 + (billable_hours - 1) * 20
print(f"Parking Duration : {hours} hours {minutes} minutes")
print(f"Billable Hours   : {billable_hours}")
print(f"Parking Fee      : ₹{fee}")


Q7. Delivery Charge & Earnings Report


n = int(input("Number of Deliveries: "))
total_distance = 0
total_earnings = 0
for i in range(n):
    distance = float(input(f"Distance {i + 1}: "))
    total_distance += distance
    if distance <= 5:
        earning = 40
    else:
        earning = 40 + (distance - 5) * 8
    total_earnings += earning
average = total_distance / n
print(f"Total Distance   : {total_distance:.2f} km")
print(f"Total Earnings   : ₹{total_earnings:.2f}")
print(f"Average Distance : {average:.2f} km")


Q8. LeetCode #7 — Reverse Integer


x = int(input("Enter integer: "))
sign = -1 if x < 0 else 1
x = abs(x)
reverse = 0
while x > 0:
    digit = x % 10
    reverse = reverse * 10 + digit
    x //= 10
reverse *= sign
print(reverse)


Q9. LeetCode #9 — Palindrome Number


x = int(input("Enter integer: "))
if x < 0:
    print(False)
else:
    original = x
    reverse = 0
    while x > 0:
        digit = x % 10
        reverse = reverse * 10 + digit
        x //= 10
    print(original == reverse)


Q10. LeetCode #412 — Fizz Buzz


N = int(input("Enter N: "))
for i in range(1, N + 1):
    if i % 3 == 0 and i % 5 == 0:
        print("FizzBuzz")
    elif i % 3 == 0:
        print("Fizz")
    elif i % 5 == 0:
        print("Buzz")
    else:
